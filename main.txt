
#include <Arduino.h>

#include "audio_manager.h"
#include "ble_manager.h"

AudioManager audioManager;

void setup()
{
    Serial.begin(115200);
    delay(1000);

    Serial.println();
    Serial.println("================================");
    Serial.println(" SIX CHANNEL AUDIO CONTROLLER");
    Serial.println("================================");

    audioManager.begin();

    initializeBLE(audioManager);
}

void loop()
{
    audioManager.processPersistence();

    delay(10);
}
