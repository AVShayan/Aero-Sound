#include <WiFi.h>
#include <WebServer.h>
#include <Wire.h>
#include <math.h>

// --- Wi-Fi Configuration ---
const char* ssid     = "TP-Link_9D2C";     
const char* password = "40266497"; 


// --- Hardware Pin Configurations ---
constexpr int SDA_PIN = 21;
constexpr int SCL_PIN = 22;
constexpr uint8_t PT2258_ADDR = 0x40;
constexpr int MAX_OUTPUT_ATTENUATION_DB = 7;

// --- Volume Variables ---
int currentPercent[2] = {50, 50}; // Index 0 = Channel 2, Index 1 = Channel 4
int targetPercent[2]  = {50, 50};
unsigned long lastRampMs[2] = {0, 0};
constexpr uint32_t RAMP_INTERVAL_MS = 4; // Smooth 4ms ramping steps

WebServer server(80);

// VERIFIED MULTI-SEGMENT AUDIO TAPER CURVE
int percentToAttenuationLog(int percent) {
  percent = constrain(percent, 0, 100);
  if (percent == 0) return 79; // Dead silent

  int attenuation = 79;

  // Segment 1: Low Volume Range (1% to 15%)
  if (percent <= 15) {
    attenuation = map(percent, 1, 15, 79, 46);
  } 
  // Segment 2: Main Listening Range (16% to 70%)
  else if (percent <= 70) {
    attenuation = map(percent, 16, 70, 45, 18);
  } 
  // Segment 3: High Power Range (71% to 100%)
  else {
    attenuation = map(percent, 71, 100, 17, MAX_OUTPUT_ATTENUATION_DB);
  }

  return constrain(attenuation, MAX_OUTPUT_ATTENUATION_DB, 79);
}

// Low-level I2C Transmission Layer
bool setChannelAttenuation(uint8_t channel, int attenuationDb) {
  attenuationDb = constrain(attenuationDb, 0, 79);
  uint8_t tens = attenuationDb / 10;
  uint8_t ones = attenuationDb % 10;
  uint8_t tenDbCommand = 0;
  uint8_t oneDbCommand = 0;

  if (channel == 2) {
    tenDbCommand = 0x40 | tens;
    oneDbCommand = 0x50 | ones;
  } else if (channel == 4) {
    tenDbCommand = 0x20 | tens;
    oneDbCommand = 0x30 | ones;
  } else {
    return false;
  }

  Wire.beginTransmission(PT2258_ADDR);
  Wire.write(tenDbCommand);
  Wire.write(oneDbCommand);
  return Wire.endTransmission() == 0;
}

void handleSetVolume() {
  if (server.hasArg("ch") && server.hasArg("val")) {
    int ch = server.arg("ch").toInt();
    int val = server.arg("val").toInt();
    if ((ch == 2 || ch == 4) && val >= 0 && val <= 100) {
      int index = (ch == 2) ? 0 : 1;
      targetPercent[index] = val; 
      server.send(200, "text/plain", "OK");
      return;
    }
  }
  server.send(400, "text/plain", "Bad Request");
}

void processVolumeRamps() {
  unsigned long now = millis();
  for (int i = 0; i < 2; i++) {
    if (currentPercent[i] != targetPercent[i]) {
      if (now - lastRampMs[i] >= RAMP_INTERVAL_MS) {
        lastRampMs[i] = now;
        currentPercent[i] += (targetPercent[i] > currentPercent[i]) ? 1 : -1;
        setChannelAttenuation((i == 0) ? 2 : 4, percentToAttenuationLog(currentPercent[i]));
      }
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(300);
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(20000); 

  Wire.beginTransmission(PT2258_ADDR);
  Wire.write(0xC0); 
  Wire.endTransmission();
  delay(5);

  setChannelAttenuation(2, percentToAttenuationLog(currentPercent[0]));
  delayMicroseconds(500);
  setChannelAttenuation(4, percentToAttenuationLog(currentPercent[1]));

  Serial.print("Connecting to Wi-Fi.");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWi-Fi Connected successfully.");
  Serial.print("Dashboard Link: http://");
  Serial.println(WiFi.localIP()); 

  server.on("/", handleRoot);
  server.on("/setvol", handleSetVolume);
  server.begin();
}

void loop() {
  server.handleClient();
  processVolumeRamps();
}
